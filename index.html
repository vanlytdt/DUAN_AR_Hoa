function phatGiongNoi(cauNoi) { // Thử phát giọng nói if ('speechSynthesis' in window && giongNoiChuan) { const msg = new SpeechSynthesisUtterance(cauNoi); msg.lang = 'vi-VN'; msg.voice = giongNoiChuan; msg.rate = 0.9; window.speechSynthesis.speak(msg); } // Luôn hiển thị text (fallback) hienThiPhuDe(cauNoi); } function hienThiPhuDe(cauNoi) { // Tạo hoặc cập nhật phụ đề let phuDe = document.getElementById('phu-de'); if (!phuDe) { phuDe = document.createElement('div'); phuDe.id = 'phu-de'; phuDe.style.cssText = ` position: fixed; bottom: 20px; left: 50%; transform: translateX(-50%); background: rgba(0, 0, 0, 0.8); color: white; padding: 10px 20px; border-radius: 8px; font-size: 14px; z-index: 10000; max-width: 90%; text-align: center; `; document.body.appendChild(phuDe); } phuDe.innerText = cauNoi; // Ẩn sau 10 giây setTimeout(() => { phuDe.style.display = 'none'; }, 10000); }
